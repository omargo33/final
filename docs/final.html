2.2 Procesamiento del texto (content)

Code 
# Función simple de limpieza de texto
def clean_text(text):
    if pd.isna(text):
        return ""
    
    text = str(text).lower()
    # Remover URLs, menciones, hashtags, ademas números y caracteres especiales
    text = re.sub(r'http\S+|www\S+', '', text)
    text = re.sub(r'@\w+', '', text)
    text = re.sub(r'#\w+', '', text)
    text = re.sub(r'\d+', '', text)
    text = re.sub(r'[^\w\s]', ' ', text)
    text = ' '.join(text.split())
    
    return text

# Aplicar limpieza
df_procesar['content_clean'] = df_procesar['content'].apply(clean_text)

# Remover stopwords en español
spanish_stopwords = set(stopwords.words('spanish'))

def remove_stopwords(text):
    words = text.split()
    return ' '.join([word for word in words if word not in spanish_stopwords and len(word) > 2])

# Crear texto procesado final
df_procesar['text_processed'] = df_procesar['content_clean'].apply(remove_stopwords)

3.4 Clasificación supervisada

Hago un bucle para entrenar el modelo, en la entrega anterior pense en que programaticamente se puede hacer un loop para ejecutar los modelos, y puedo comparar de mejor manera al convertirla en una funcion de facil uso

Se usa OVR para que “sea uno contra el resto” en la etiquetacion de clases en multi capa.

Code 
# Seteo los modelos en un arreglo.
models = {
    'logReg': LogisticRegression(random_state=42, max_iter=1000),
    'ramFor': RandomForestClassifier(n_estimators=100, random_state=42),
    'Bayes': MultinomialNB(),
    'SVC': SVC(probability=True, random_state=42)
}

def entrenamiento_evaluacion_masiva(X_train, X_test, y_train, y_test, models, paso_nombre):
    results = {}
    for name, model in models.items():
        print(f"\nEntrenando {name}...")
        
        model.fit(X_train, y_train)
        
        y_pred = model.predict(X_test)        

        accuracy = accuracy_score(y_test, y_pred)
        precision = precision_score(y_test, y_pred, average='weighted')
        recall = recall_score(y_test, y_pred, average='weighted')
        f1 = f1_score(y_test, y_pred, average='weighted')
        
        ## Para multi capas o clases se revisa la documentacion en: https://scikit-learn.org/stable/modules/generated/sklearn.metrics.roc_auc_score.html para el roc_auc_score
        y_pred_proba = model.predict_proba(X_test)
        if len(np.unique(y_test)) == 2:
            roc_auc = roc_auc_score(y_test, y_pred_proba[:, 1])
        else:
            ## ovr que hace con uno (contra el resto)
            roc_auc = roc_auc_score(y_test, y_pred_proba, multi_class='ovr')
        
        results[name] = {
            'model': model,
            'accuracy': accuracy,
            'precision': precision,
            'recall': recall,
            'f1_score': f1,
            'roc_auc': roc_auc,
            'y_pred': y_pred, 
            'y_pred_proba': y_pred_proba
        }
        
        print(f"- Accuracy: {accuracy:.4f}")
        print(f"- Precision: {precision:.4f}")
        print(f"- Recall: {recall:.4f}")
        print(f"- F1-Score: {f1:.4f}")
        print(f"- ROC-AUC: {roc_auc:.4f}")
    
    return results

## Los resultados deben ser presentados en bucle

def analizo_modelo_por_metrica(results, y_test, paso_nombre, class_names=None):
    metricas_df = pd.DataFrame({
        'Model': list(results.keys()),
        'Accuracy': [results[model]['accuracy'] for model in results.keys()],
        'Precision': [results[model]['precision'] for model in results.keys()],
        'Recall': [results[model]['recall'] for model in results.keys()],
        'F1-Score': [results[model]['f1_score'] for model in results.keys()],
        'ROC-AUC': [results[model]['roc_auc'] for model in results.keys()]
    }).round(4)
    
    # Encontrar mejor modelo, tomado de: https://pandas.pydata.org/docs/reference/api/pandas.DataFrame.loc.html
    best_model_name = metricas_df.loc[metricas_df['F1-Score'].idxmax(), 'Model']
    
    print(f"\nMEJOR MODELO: {best_model_name} {paso_nombre}")
    
    print(f"Valor del F1-Score: {metricas_df.loc[metricas_df['Model'] == best_model_name, 'F1-Score'].iloc[0]}")
    
    return metricas_df, best_model_name

def crear_pipeline_regresion(modelo_regresion):
    pipeline = Pipeline([
        ('preprocessor', preprocessor_simple),  # TF-IDF ya definido
        ('regressor', modelo_regresion)
    ])
    return pipeline

## Arreglo de modelos para poder ejecutarlos en Loop y Luego poder comparar.
pipelines_regresion = {
    'Linear Regression': crear_pipeline_regresion(LinearRegression()),
    'Ridge Regression': crear_pipeline_regresion(Ridge(alpha=1.0, random_state=42)),
    'Lasso Regression': crear_pipeline_regresion(Lasso(alpha=0.01, random_state=42, max_iter=2000)),
    'Random Forest': crear_pipeline_regresion(RandomForestRegressor(n_estimators=100, random_state=42)),
    'Gradient Boosting': crear_pipeline_regresion(GradientBoostingRegressor(n_estimators=100, random_state=42)),
    'SVR': crear_pipeline_regresion(SVR(kernel='rbf', C=1.0))
}

def entrenar_evaluar_regresion_pipeline(X_train, X_test, y_train, y_test, pipelines):
    resultados = {}
    
    for nombre, pipeline in pipelines.items():
        print(f"\nEntrenando {nombre}")
        
        start_time = time.time()        
        pipeline.fit(X_train, y_train)
        
        tiempo_entrenamiento = time.time() - start_time
        
        y_pred_train = pipeline.predict(X_train)
        y_pred_test = pipeline.predict(X_test)
        
        mae_train = mean_absolute_error(y_train, y_pred_train)
        rmse_train = np.sqrt(mean_squared_error(y_train, y_pred_train))
        r2_train = r2_score(y_train, y_pred_train)
        
        mae_test = mean_absolute_error(y_test, y_pred_test)
        rmse_test = np.sqrt(mean_squared_error(y_test, y_pred_test))
        r2_test = r2_score(y_test, y_pred_test)
        
        resultados[nombre] = {
            'pipeline': pipeline,
            'modelo': pipeline.named_steps['regressor'], 
            'tiempo_entrenamiento': tiempo_entrenamiento,
            'mae_train': mae_train,
            'rmse_train': rmse_train,
            'r2_train': r2_train,
            'mae_test': mae_test,
            'rmse_test': rmse_test,
            'r2_test': r2_test,
            'y_pred_train': y_pred_train,
            'y_pred_test': y_pred_test
        }
        
        print(f"  - Tiempo entrenamiento: {tiempo_entrenamiento:.3f}s")
        print(f"  - MAE (test): {mae_test:.4f}")
        print(f"  - RMSE (test): {rmse_test:.4f}")
        print(f"  - R2 (test): {r2_test:.4f}")
    
    return resultados

Método del Codo (Elbow Method)

Code 
k_range = range(2, 11)
inertias = []
silhouette_scores = []
X_clustering = X_final.toarray()  # Convertir sparse matrix a dense para clustering

# Probar diferentes números de clusters entre el 2 y el 10
for k in k_range:
    print(f"Probando k={k}...")
    kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
    kmeans.fit(X_clustering)
    
    inertias.append(kmeans.inertia_)
    silhouette_avg = silhouette_score(X_clustering, kmeans.labels_)
    silhouette_scores.append(silhouette_avg)
    
    print(f"- Inercia: {kmeans.inertia_:.2f} - Silhouette Score: {silhouette_avg:.4f}")

best_k_silhouette = k_range[np.argmax(silhouette_scores)]
best_silhouette = max(silhouette_scores)


## Dimension del grafico
